// Aspect of Death (the Eternal). Goal, Sorrow: every point of damage the Eternal takes in combat goes into a
// ledger that healing does not undo. Each tier has its own goal: 30% / 50% / 70% of maximum HP (the table in
// ARCANA_Aspect_Common.txt, in thousandths). The same ledger is The Final Toll's damage.
//
// DBs:
//   DB_ARCANA_SorrowLedger(Eternal, Damage)   the damage taken this fight. Deleted by the reset when she leaves
//                                             combat.
// Reads: DB_ARCANA_Aspect_InCombat, DB_ARCANA_AspectReached (ARCANA_Aspect_Common.txt), DB_ARCANA_CardFamily.
// Hooks: PROC_ARCANA_Aspect_OnAwakened, PROC_ARCANA_Aspect_OnTier, PROC_ARCANA_Aspect_Reset.

//REGION Ledger and goal
IF
AttackedBy((CHARACTER)_Eternal, _, _, _, _Amount, _, _)
AND
_Amount > 0
AND
HasPassive(_Eternal, "Arcana_Passive_Aspect_Death_I", 1)
AND
IsInCombat(_Eternal, 1)
THEN
PROC_ARCANA_Death_Suffer(_Eternal, _Amount);

// Blood Offering: AttackedBy does not fire for self-damage (tested 2026-10-02), so the cast credits the same
// MaxHP/10 that the card's SpellProperties takes.
IF
CastedSpell((CHARACTER)_Eternal, _Spell, _, _, _)
AND
DB_ARCANA_CardFamily(_Spell, "Shout_Arcana_Card_Spell_BloodOffering")
AND
HasPassive(_Eternal, "Arcana_Passive_Aspect_Death_I", 1)
AND
IsInCombat(_Eternal, 1)
AND
GetMaxHitpoints(_Eternal, _Max)
AND
IntegerDivide(_Max, 10, _Cost)
AND
_Cost > 0
THEN
PROC_ARCANA_Death_Suffer(_Eternal, _Cost);

PROC
PROC_ARCANA_Death_Suffer((CHARACTER)_Eternal, (INTEGER)_Amount)
AND
NOT DB_ARCANA_SorrowLedger(_Eternal, _)
THEN
DB_ARCANA_SorrowLedger(_Eternal, 0);

PROC
PROC_ARCANA_Death_Suffer((CHARACTER)_Eternal, (INTEGER)_Amount)
AND
DB_ARCANA_SorrowLedger(_Eternal, _Old)
AND
IntegerSum(_Old, _Amount, _Sum)
THEN
NOT DB_ARCANA_SorrowLedger(_Eternal, _Old);
DB_ARCANA_SorrowLedger(_Eternal, _Sum);
PROC_ARCANA_Death_Measure(_Eternal, _Sum);

// the loss in thousandths of maximum HP (300 = 30%)
PROC
PROC_ARCANA_Death_Measure((CHARACTER)_Eternal, (INTEGER)_Sum)
AND
GetMaxHitpoints(_Eternal, _Max)
AND
IntegerProduct(_Sum, 1000, _Sum1000)
AND
IntegerDivide(_Sum1000, _Max, _PerMille)
THEN
PROC_ARCANA_Aspect_Progress(_Eternal, "DEATH", _PerMille);
//END_REGION

//REGION Tier I (Sorrow) and Tier II (Mourning)
PROC
PROC_ARCANA_Aspect_OnAwakened((CHARACTER)_Eternal, (STRING)_Aspect)
AND
_Aspect == "DEATH"
THEN
ApplyStatus(_Eternal, "DARKNESS_RISE_AURA", -1.0, 1, _Eternal);

PROC
PROC_ARCANA_Aspect_OnTier((CHARACTER)_Eternal, (STRING)_Aspect, (INTEGER)_Tier)
AND
_Aspect == "DEATH"
AND
_Tier == 2
THEN
ApplyStatus(_Eternal, "ETERNAL_UNDYING", -1.0, 1, _Eternal);
ApplyStatus(_Eternal, "ETERNAL_DREAD_AURA", -1.0, 1, _Eternal);
//END_REGION

//REGION Tier III: Harvest
// DB_ARCANA_AspectReached keeps no Aspect: without the passive, any subclass at Tier III would gain Harvest.
IF
Died(_Dead)
AND
DB_ARCANA_Aspect_InCombat(_Eternal, _)
AND
DB_ARCANA_AspectReached(_Eternal, 3)
AND
HasPassive(_Eternal, "Arcana_Passive_Aspect_Death_III", 1)
AND
HasActiveStatus(_Eternal, "HARVEST_SPENT", 0)
AND
IsEnemy(_Eternal, _Dead, 1)
AND
GetDistanceTo(_Eternal, _Dead, _Dist)
AND
_Dist <= 9.0
THEN
ApplyStatus(_Eternal, "HARVEST_ACTION", 6.0, 1, _Eternal);
ApplyStatus(_Eternal, "HARVEST_SPENT", 6.0, 1, _Eternal);
//END_REGION

//REGION The Final Toll: damage = the ledger, capped at half the maximum HP
IF
StatusApplied((CHARACTER)_Foe, "FINAL_TOLL_TECH", (CHARACTER)_Eternal, _)
AND
DB_ARCANA_SorrowLedger(_Eternal, _Ledger)
AND
GetMaxHitpoints(_Eternal, _Max)
AND
IntegerDivide(_Max, 2, _Cap)
AND
IntegerMin(_Ledger, _Cap, _Damage)
AND
_Damage > 0
THEN
ApplyDamage(_Foe, _Damage, "Necrotic", _Eternal);
//END_REGION

//REGION Cleanup
PROC
PROC_ARCANA_Aspect_Reset((CHARACTER)_Char)
AND
DB_ARCANA_SorrowLedger(_Char, _N)
THEN
NOT DB_ARCANA_SorrowLedger(_Char, _N);

PROC
PROC_ARCANA_Aspect_Reset((CHARACTER)_Char)
AND
HasPassive(_Char, "Arcana_Passive_Aspect_Death_I", 1)
THEN
RemoveStatus(_Char, "DARKNESS_RISE_AURA", _Char);

PROC
PROC_ARCANA_Aspect_Reset((CHARACTER)_Char)
AND
HasPassive(_Char, "Arcana_Passive_Aspect_Death_II", 1)
THEN
RemoveStatus(_Char, "ETERNAL_DREAD_AURA", _Char);
RemoveStatus(_Char, "ETERNAL_UNDYING", _Char);
//END_REGION
EXITSECTION

ENDEXITSECTION
