// The Kingmaker's Wheel (the Aspect of the Wheel of Fortune). The ledger: owner, creature, position, side (1 ally /
// 0 enemy, fixed at entry), entry order. The Wheel turns at the owner's turn start; two Kingmakers keep separate
// Wheels (SP113). Each region title names the feature and the level or tier that brings it.
// Positions run Rising -> Reigning -> Falling -> Fallen -> Rising. An enemy entering Fallen near a Reigning ally
// pays the Crossing (Force damage, once per enemy per round); each Crossing is an Awakening step (a lone enemy
// counts twice). The crown goes to the ally Reigning longest; the Court's offices last until the next Long Rest.
// Wheel of Ixion binds through KINGMAKER_BIND_ENEMY_TECH like Dethrone. During The Wheel Stops the Kingmaker's own
// cards still move creatures (SP112).
//
// DBs (tables, rebuilt by PROC_ARCANA_Kingmaker_Init on a new game and on every load):
//   DB_ARCANA_Kingmaker_Next(From, To)  DB_ARCANA_Kingmaker_Opposite(From, To)  DB_ARCANA_Kingmaker_PositionStatus(Position, Status)
//   DB_ARCANA_Kingmaker_Office(Status, Office)  and the positions where a step applies: DB_ARCANA_Kingmaker_HealStep,
//   DB_ARCANA_Kingmaker_LawStep, DB_ARCANA_Kingmaker_ForeseenStep (Position)
// DBs (the Wheel; deleted when a creature dies or is cut, and at the end of the owner's fight):
//   DB_ARCANA_Kingmaker_OnWheel(Owner, Creature, Position, Side, Order)   the ledger
//   DB_ARCANA_Kingmaker_Counter(Owner, Last)       last entry order handed out
//   DB_ARCANA_Kingmaker_Best(Owner, King, Order)   the crown's holder
//   DB_ARCANA_Kingmaker_Crossed(Owner, Enemy)      already crossed this round (cleared at the owner's turn start)
//   DB_ARCANA_Kingmaker_Held(Owner, Ally)          Tier III: an ally kept Falling one more turn
//   DB_ARCANA_Kingmaker_Condemned(Owner, Enemy)    Master of Laws: sentenced for the next turn of the Wheel
//   DB_ARCANA_Kingmaker_Stopped(Owner, Turns)      The Wheel Stops: turns left
//   DB_ARCANA_Kingmaker_Enemies(Owner, N)          enemies on the Wheel, counted at each Crossing
//   DB_ARCANA_Kingmaker_UsurpFirst(Owner, Creature, Position, Side, Order)   Usurp's first target, waiting for the second
// DBs (transient: written and consumed in the same frame):
//   DB_ARCANA_Kingmaker_Moving, DB_ARCANA_Kingmaker_Pending, DB_ARCANA_Kingmaker_LawPending,
//   DB_ARCANA_Kingmaker_Inverting, DB_ARCANA_Kingmaker_Usurping, DB_ARCANA_Kingmaker_NewFallen,
//   DB_ARCANA_Kingmaker_NewReigning, DB_ARCANA_Kingmaker_CrossPending, DB_ARCANA_Kingmaker_NemesisPending,
//   DB_ARCANA_Kingmaker_NemesisMoved, DB_ARCANA_Kingmaker_Heir
// DBs (the Court; deleted at the Long Rest):
//   DB_ARCANA_Kingmaker_Court(Owner, Holder, Office)   DB_ARCANA_Kingmaker_Seats(Owner, Taken)
// Reads: DB_ARCANA_Aspect_InCombat, DB_ARCANA_AspectReached (ARCANA_Aspect_Common.txt). Calls PROC_ARCANA_Aspect_Step.

//REGION The Court (level 5) - offices until the next Long Rest; seats = Charisma modifier, at least 1
IF
StatusApplied((CHARACTER)_Holder, _Status, _Owner, _)
AND
DB_ARCANA_Kingmaker_Office(_Status, _Office)
THEN
DB_ARCANA_Kingmaker_Court(_Owner, _Holder, _Office);
ApplyStatus(_Holder, "COURT_HOLDER", -1.0, 1, _Owner);
PROC_ARCANA_Kingmaker_TakeSeat((CHARACTER)_Owner);

PROC
PROC_ARCANA_Kingmaker_TakeSeat((CHARACTER)_Owner)
AND
NOT DB_ARCANA_Kingmaker_Seats(_Owner, _)
THEN
DB_ARCANA_Kingmaker_Seats(_Owner, 0);

PROC
PROC_ARCANA_Kingmaker_TakeSeat((CHARACTER)_Owner)
AND
DB_ARCANA_Kingmaker_Seats(_Owner, _Old)
AND
IntegerSum(_Old, 1, _New)
THEN
NOT DB_ARCANA_Kingmaker_Seats(_Owner, _Old);
DB_ARCANA_Kingmaker_Seats(_Owner, _New);

// full when the seats taken reach the Charisma modifier (at least 1): the Convene the Court options then can't be cast
PROC
PROC_ARCANA_Kingmaker_TakeSeat((CHARACTER)_Owner)
AND
DB_ARCANA_Kingmaker_Seats(_Owner, _N)
AND
GetAbility(_Owner, "Charisma", _Score)
AND
IntegerSubtract(_Score, 10, _X)
AND
IntegerDivide(_X, 2, _Mod)
AND
_Mod < 1
AND
_N >= 1
THEN
ApplyStatus(_Owner, "COURT_FULL", -1.0, 1, _Owner);

PROC
PROC_ARCANA_Kingmaker_TakeSeat((CHARACTER)_Owner)
AND
DB_ARCANA_Kingmaker_Seats(_Owner, _N)
AND
GetAbility(_Owner, "Charisma", _Score)
AND
IntegerSubtract(_Score, 10, _X)
AND
IntegerDivide(_X, 2, _Mod)
AND
_Mod >= 1
AND
_N >= _Mod
THEN
ApplyStatus(_Owner, "COURT_FULL", -1.0, 1, _Owner);

// an office that ends (death, Long Rest) leaves the Court; its seat stays taken until the Long Rest
IF
StatusRemoved((CHARACTER)_Holder, _Status, _, _)
AND
DB_ARCANA_Kingmaker_Office(_Status, _Office)
AND
DB_ARCANA_Kingmaker_Court(_Owner, _Holder, _Office)
THEN
NOT DB_ARCANA_Kingmaker_Court(_Owner, _Holder, _Office);
RemoveStatus(_Holder, "COURT_HOLDER", _Owner);
RemoveStatus(_Holder, "KINGMAKER_HAND_CROWN", _Owner);
PROC_ARCANA_Kingmaker_Crown(_Owner);

IF
Died(_Holder)
AND
DB_ARCANA_Kingmaker_Court(_Owner, _Holder, _Office)
AND
DB_ARCANA_Kingmaker_Office(_Status, _Office)
THEN
RemoveStatus(_Holder, _Status, _Owner);

IF
LongRestFinished()
AND
DB_ARCANA_Kingmaker_Court(_Owner, _Holder, _Office)
AND
DB_ARCANA_Kingmaker_Office(_Status, _Office)
THEN
RemoveStatus(_Holder, _Status, _Owner);

IF
LongRestFinished()
AND
DB_ARCANA_Kingmaker_Seats(_Owner, _N)
THEN
RemoveStatus(_Owner, "KINGMAKER_COIN_SPENT", _Owner);

IF
LongRestFinished()
AND
DB_ARCANA_Kingmaker_Seats(_Owner, _N)
THEN
NOT DB_ARCANA_Kingmaker_Seats(_Owner, _N);
RemoveStatus(_Owner, "COURT_FULL", _Owner);
//END_REGION

//REGION Master of Coin (level 9) - once per Long Rest, its hit on a Fallen enemy gives the Kingmaker a spell slot back
IF
StatusApplied((CHARACTER)_Enemy, "KINGMAKER_COIN_TECH", (CHARACTER)_Holder, _)
AND
DB_ARCANA_Kingmaker_Court(_Owner, _Holder, "MASTER_OF_COIN")
AND
HasActiveStatus(_Owner, "KINGMAKER_COIN_SPENT", 0)
THEN
ApplyStatus(_Owner, "KINGMAKER_COIN_SPENT", -1.0, 1, _Owner);
ApplyStatus(_Owner, "KINGMAKER_COIN_RESTORE", 0.0, 1, _Owner);
//END_REGION

//REGION Bind - Coronation (ally, Rising) and Dethrone (enemy, Falling); with Tyche (Tier I) one position ahead
IF
StatusApplied((CHARACTER)_Creature, "KINGMAKER_BIND_ALLY_TECH", (CHARACTER)_Owner, _)
AND
NOT DB_ARCANA_AspectReached(_Owner, 1)
THEN
PROC_ARCANA_Kingmaker_Bind(_Owner, _Creature, "RISING", 1);

IF
StatusApplied((CHARACTER)_Creature, "KINGMAKER_BIND_ALLY_TECH", (CHARACTER)_Owner, _)
AND
DB_ARCANA_AspectReached(_Owner, 1)
THEN
PROC_ARCANA_Kingmaker_Bind(_Owner, _Creature, "REIGNING", 1);

IF
StatusApplied((CHARACTER)_Creature, "KINGMAKER_BIND_ENEMY_TECH", (CHARACTER)_Owner, _)
AND
NOT DB_ARCANA_AspectReached(_Owner, 1)
THEN
PROC_ARCANA_Kingmaker_Bind(_Owner, _Creature, "FALLING", 0);

IF
StatusApplied((CHARACTER)_Creature, "KINGMAKER_BIND_ENEMY_TECH", (CHARACTER)_Owner, _)
AND
DB_ARCANA_AspectReached(_Owner, 1)
THEN
PROC_ARCANA_Kingmaker_Bind(_Owner, _Creature, "FALLEN", 0);

PROC
PROC_ARCANA_Kingmaker_Bind((GUIDSTRING)_Owner, (CHARACTER)_Creature, (STRING)_Position, (INTEGER)_Side)
AND
NOT DB_ARCANA_Kingmaker_OnWheel(_, _Creature, _, _, _)
AND
NOT DB_ARCANA_Kingmaker_Counter(_Owner, _)
THEN
DB_ARCANA_Kingmaker_Counter(_Owner, 0);

PROC
PROC_ARCANA_Kingmaker_Bind((GUIDSTRING)_Owner, (CHARACTER)_Creature, (STRING)_Position, (INTEGER)_Side)
AND
NOT DB_ARCANA_Kingmaker_OnWheel(_, _Creature, _, _, _)
AND
DB_ARCANA_Kingmaker_Counter(_Owner, _Old)
AND
IntegerSum(_Old, 1, _New)
AND
DB_ARCANA_Kingmaker_PositionStatus(_Position, _Status)
THEN
NOT DB_ARCANA_Kingmaker_Counter(_Owner, _Old);
DB_ARCANA_Kingmaker_Counter(_Owner, _New);
DB_ARCANA_Kingmaker_OnWheel(_Owner, _Creature, _Position, _Side, _New);
ApplyStatus(_Creature, _Status, -1.0, 1, _Owner);
PROC_ARCANA_Kingmaker_Fell(_Owner, _Creature, _Position, _Side);
PROC_ARCANA_Kingmaker_Crown(_Owner);
PROC_ARCANA_Kingmaker_Settle(_Owner);

PROC
PROC_ARCANA_Kingmaker_Bind((GUIDSTRING)_Owner, (CHARACTER)_Creature, (STRING)_Position, (INTEGER)_Side)
AND
DB_ARCANA_Kingmaker_Stopped(_Owner, _)
AND
DB_ARCANA_Kingmaker_OnWheel(_Owner, _Creature, _, _, _)
THEN
ApplyStatus(_Creature, "KINGMAKER_WHEEL_STOPPED", -1.0, 1, _Owner);
//END_REGION

//REGION Move one step (collected first, so the ledger is never changed while it is read)
PROC
PROC_ARCANA_Kingmaker_Move((GUIDSTRING)_Owner, (CHARACTER)_Creature)
AND
DB_ARCANA_Kingmaker_OnWheel(_Owner, _Creature, _Position, _Side, _Order)
THEN
DB_ARCANA_Kingmaker_Moving(_Owner, _Creature, _Position, _Side, _Order);

// Tier III (Aspect of Anticipation): an ally that already spent one more turn Falling goes on to Rising
PROC
PROC_ARCANA_Kingmaker_Move((GUIDSTRING)_Owner, (CHARACTER)_Creature)
AND
DB_ARCANA_Kingmaker_Moving(_Owner, _Creature, "FALLING", 1, _Order)
AND
DB_ARCANA_Kingmaker_Held(_Owner, _Creature)
THEN
NOT DB_ARCANA_Kingmaker_Moving(_Owner, _Creature, "FALLING", 1, _Order);
NOT DB_ARCANA_Kingmaker_Held(_Owner, _Creature);
NOT DB_ARCANA_Kingmaker_OnWheel(_Owner, _Creature, "FALLING", 1, _Order);
DB_ARCANA_Kingmaker_OnWheel(_Owner, _Creature, "RISING", 1, _Order);
RemoveStatus(_Creature, "KINGMAKER_FALLING", _Owner);
ApplyStatus(_Creature, "KINGMAKER_RISING", -1.0, 1, _Owner);

PROC
PROC_ARCANA_Kingmaker_Move((GUIDSTRING)_Owner, (CHARACTER)_Creature)
AND
DB_ARCANA_Kingmaker_Moving(_Owner, _Creature, _Position, _Side, _Order)
AND
DB_ARCANA_Kingmaker_Next(_Position, _NextPosition)
AND
DB_ARCANA_Kingmaker_PositionStatus(_Position, _OldStatus)
AND
DB_ARCANA_Kingmaker_PositionStatus(_NextPosition, _NewStatus)
THEN
NOT DB_ARCANA_Kingmaker_Moving(_Owner, _Creature, _Position, _Side, _Order);
NOT DB_ARCANA_Kingmaker_OnWheel(_Owner, _Creature, _Position, _Side, _Order);
DB_ARCANA_Kingmaker_OnWheel(_Owner, _Creature, _NextPosition, _Side, _Order);
RemoveStatus(_Creature, _OldStatus, _Owner);
ApplyStatus(_Creature, _NewStatus, -1.0, 1, _Owner);
PROC_ARCANA_Kingmaker_Fell(_Owner, _Creature, _NextPosition, _Side);

// an enemy that enters Fallen waits for the Crossing check, made after every creature has moved
PROC
PROC_ARCANA_Kingmaker_Fell((GUIDSTRING)_Owner, (CHARACTER)_Creature, (STRING)_Position, (INTEGER)_Side)
AND
_Position == "FALLEN"
AND
_Side == 0
THEN
DB_ARCANA_Kingmaker_NewFallen(_Owner, _Creature);

PROC
PROC_ARCANA_Kingmaker_Fell((GUIDSTRING)_Owner, (CHARACTER)_Creature, (STRING)_Position, (INTEGER)_Side)
AND
_Position == "REIGNING"
AND
_Side == 0
THEN
DB_ARCANA_Kingmaker_NewReigning(_Owner, _Creature);

// Tier III: an ally that would enter Fallen (any cause) stays Falling one more turn instead
PROC
PROC_ARCANA_Kingmaker_Fell((GUIDSTRING)_Owner, (CHARACTER)_Creature, (STRING)_Position, (INTEGER)_Side)
AND
_Position == "FALLEN"
AND
_Side == 1
AND
HasActiveStatus(_Owner, "ASPECT_AWAKENED_WHEEL_III", 1)
AND
DB_ARCANA_Kingmaker_OnWheel(_Owner, _Creature, "FALLEN", 1, _Order)
THEN
NOT DB_ARCANA_Kingmaker_OnWheel(_Owner, _Creature, "FALLEN", 1, _Order);
DB_ARCANA_Kingmaker_OnWheel(_Owner, _Creature, "FALLING", 1, _Order);
RemoveStatus(_Creature, "KINGMAKER_FALLEN", _Owner);
ApplyStatus(_Creature, "KINGMAKER_FALLING", -1.0, 1, _Owner);
DB_ARCANA_Kingmaker_Held(_Owner, _Creature);
//END_REGION

//REGION The Wheel turns at the owner's turn start (a new round: every enemy may cross again)
IF
TurnStarted(_Owner)
AND
DB_ARCANA_Kingmaker_Crossed(_Owner, _Enemy)
THEN
NOT DB_ARCANA_Kingmaker_Crossed(_Owner, _Enemy);

IF
TurnStarted(_Owner)
AND
NOT DB_ARCANA_Kingmaker_Stopped(_Owner, _)
AND
DB_ARCANA_Kingmaker_OnWheel(_Owner, _Creature, _, _, _)
THEN
DB_ARCANA_Kingmaker_Pending(_Owner, _Creature);

IF
TurnStarted(_Owner)
AND
DB_ARCANA_Kingmaker_Pending(_Owner, _Creature)
THEN
NOT DB_ARCANA_Kingmaker_Pending(_Owner, _Creature);
PROC_ARCANA_Kingmaker_Move(_Owner, _Creature);

// Master of Laws: an enemy that attacked the King and now Reigns (came from Rising) or Falls (came from Reigning) takes one
// more step - it skips its reign, or drops to Fallen; from Falling or Fallen nothing more
IF
TurnStarted(_Owner)
AND
NOT DB_ARCANA_Kingmaker_Stopped(_Owner, _)
AND
DB_ARCANA_Kingmaker_Condemned(_Owner, _Enemy)
AND
DB_ARCANA_Kingmaker_OnWheel(_Owner, _Enemy, _Position, 0, _)
AND
DB_ARCANA_Kingmaker_LawStep(_Position)
THEN
DB_ARCANA_Kingmaker_LawPending(_Owner, _Enemy);

IF
TurnStarted(_Owner)
AND
DB_ARCANA_Kingmaker_LawPending(_Owner, _Enemy)
THEN
NOT DB_ARCANA_Kingmaker_LawPending(_Owner, _Enemy);
PROC_ARCANA_Kingmaker_Move(_Owner, _Enemy);

IF
TurnStarted(_Owner)
AND
DB_ARCANA_Kingmaker_Condemned(_Owner, _Enemy)
THEN
NOT DB_ARCANA_Kingmaker_Condemned(_Owner, _Enemy);

IF
TurnStarted(_Owner)
AND
DB_ARCANA_Kingmaker_UsurpFirst(_Owner, _Creature, _Position, _Side, _Order)
THEN
NOT DB_ARCANA_Kingmaker_UsurpFirst(_Owner, _Creature, _Position, _Side, _Order);

// The Wheel Stops: every Fallen enemy on the stopped Wheel is checked for its Crossing again (once per round still holds)
IF
TurnStarted(_Owner)
AND
DB_ARCANA_Kingmaker_Stopped(_Owner, _)
AND
DB_ARCANA_Kingmaker_OnWheel(_Owner, _Enemy, "FALLEN", 0, _)
THEN
DB_ARCANA_Kingmaker_NewFallen(_Owner, _Enemy);

IF
TurnStarted(_Owner)
AND
DB_ARCANA_Kingmaker_Counter(_Owner, _)
THEN
PROC_ARCANA_Kingmaker_Crown(_Owner);
PROC_ARCANA_Kingmaker_Settle(_Owner);

IF
TurnStarted(_Owner)
AND
DB_ARCANA_Kingmaker_Stopped(_Owner, _Turns)
AND
IntegerSubtract(_Turns, 1, _Left)
THEN
NOT DB_ARCANA_Kingmaker_Stopped(_Owner, _Turns);
PROC_ARCANA_Kingmaker_StopTick(_Owner, _Left);

PROC
PROC_ARCANA_Kingmaker_StopTick((GUIDSTRING)_Owner, (INTEGER)_Left)
AND
_Left > 0
THEN
DB_ARCANA_Kingmaker_Stopped(_Owner, _Left);

PROC
PROC_ARCANA_Kingmaker_StopTick((GUIDSTRING)_Owner, (INTEGER)_Left)
AND
_Left < 1
AND
DB_ARCANA_Kingmaker_OnWheel(_Owner, _Creature, _, _, _)
THEN
RemoveStatus(_Creature, "KINGMAKER_WHEEL_STOPPED", _Owner);

IF
TurnStarted(_Owner)
AND
NOT DB_ARCANA_Kingmaker_Counter(_Owner, _)
AND
DB_ARCANA_Kingmaker_Court(_Owner, _, "KINGS_HAND")
THEN
PROC_ARCANA_Kingmaker_Crown(_Owner);
//END_REGION

//REGION Peripeteia (level 5) - every creature on the Wheel within 18 m of the Kingmaker to the opposite position, at once
IF
StatusApplied(_Owner, "KINGMAKER_INVERT_TECH", _, _)
AND
DB_ARCANA_Kingmaker_OnWheel(_Owner, _Creature, _Position, _Side, _Order)
AND
GetDistanceTo(_Owner, _Creature, _Dist)
AND
_Dist <= 18.0
THEN
DB_ARCANA_Kingmaker_Inverting(_Owner, _Creature, _Position, _Side, _Order);

IF
StatusApplied(_Owner, "KINGMAKER_INVERT_TECH", _, _)
AND
DB_ARCANA_Kingmaker_Inverting(_Owner, _Creature, _Position, _Side, _Order)
AND
DB_ARCANA_Kingmaker_Opposite(_Position, _NewPosition)
AND
DB_ARCANA_Kingmaker_PositionStatus(_Position, _OldStatus)
AND
DB_ARCANA_Kingmaker_PositionStatus(_NewPosition, _NewStatus)
THEN
NOT DB_ARCANA_Kingmaker_Inverting(_Owner, _Creature, _Position, _Side, _Order);
NOT DB_ARCANA_Kingmaker_OnWheel(_Owner, _Creature, _Position, _Side, _Order);
DB_ARCANA_Kingmaker_OnWheel(_Owner, _Creature, _NewPosition, _Side, _Order);
RemoveStatus(_Creature, _OldStatus, _Owner);
ApplyStatus(_Creature, _NewStatus, -1.0, 1, _Owner);
PROC_ARCANA_Kingmaker_Fell(_Owner, _Creature, _NewPosition, _Side);

IF
StatusApplied(_Owner, "KINGMAKER_INVERT_TECH", _, _)
AND
DB_ARCANA_Kingmaker_Counter(_Owner, _)
THEN
PROC_ARCANA_Kingmaker_Crown(_Owner);
PROC_ARCANA_Kingmaker_Settle(_Owner);
//END_REGION

//REGION Usurp (level 7) - an ally and an enemy on the Wheel trade positions (two targets: the first waits for the second)
IF
StatusApplied((CHARACTER)_Creature, "KINGMAKER_USURP_TECH", _Owner, _)
AND
NOT DB_ARCANA_Kingmaker_UsurpFirst(_Owner, _, _, _, _)
AND
DB_ARCANA_Kingmaker_OnWheel(_Owner, _Creature, _Position, _Side, _Order)
THEN
DB_ARCANA_Kingmaker_UsurpFirst(_Owner, _Creature, _Position, _Side, _Order);

IF
StatusApplied((CHARACTER)_Creature, "KINGMAKER_USURP_TECH", _Owner, _)
AND
DB_ARCANA_Kingmaker_UsurpFirst(_Owner, _First, _FirstPosition, _FirstSide, _FirstOrder)
AND
_First != _Creature
AND
DB_ARCANA_Kingmaker_OnWheel(_Owner, _Creature, _Position, _Side, _Order)
AND
_Side != _FirstSide
THEN
DB_ARCANA_Kingmaker_Usurping(_Owner, _First, _FirstPosition, _FirstSide, _FirstOrder, _Creature, _Position, _Side, _Order);

// the second target always closes the pair (two allies or two enemies don't trade)
IF
StatusApplied((CHARACTER)_Creature, "KINGMAKER_USURP_TECH", _Owner, _)
AND
DB_ARCANA_Kingmaker_UsurpFirst(_Owner, _First, _FirstPosition, _FirstSide, _FirstOrder)
AND
_First != _Creature
THEN
NOT DB_ARCANA_Kingmaker_UsurpFirst(_Owner, _First, _FirstPosition, _FirstSide, _FirstOrder);

IF
StatusApplied((CHARACTER)_Creature, "KINGMAKER_USURP_TECH", _Owner, _)
AND
DB_ARCANA_Kingmaker_Usurping(_Owner, _A, _PositionA, _SideA, _OrderA, _B, _PositionB, _SideB, _OrderB)
AND
DB_ARCANA_Kingmaker_PositionStatus(_PositionA, _StatusA)
AND
DB_ARCANA_Kingmaker_PositionStatus(_PositionB, _StatusB)
THEN
