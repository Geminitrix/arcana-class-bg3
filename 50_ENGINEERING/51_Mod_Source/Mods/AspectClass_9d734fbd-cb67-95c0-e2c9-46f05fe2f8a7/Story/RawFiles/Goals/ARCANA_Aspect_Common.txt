// Aspect Awakening, the part every Aspect shares. Each tier has its own goal; the subclass goal reports progress
// (PROC_ARCANA_Aspect_Progress, or _Step for a plain count) and here each tier reached awakens, lowest first, if
// the level unlocks it (the tier passive: levels 3 / 7 / 12). Tiers add up and last until the end of the fight.
// Tier I also applies the technical status ASPECT_AWAKENED_<X> (the Aspect card, the VFX), and each tier applies
// its own ASPECT_AWAKENED_<X>_I/II/III, which carries the bonuses. Hooks for the subclass goals:
// PROC_ARCANA_Aspect_OnAwakened(_Char, _Aspect) at Tier I and PROC_ARCANA_Aspect_OnTier(_Char, _Aspect, _Tier);
// PROC_ARCANA_Aspect_Reset(_Char) when a character's fight ends. Each Aspect's goal and effects live in
// ARCANA_Aspect_<Arcanum>.txt.
//
// DBs (tables, rebuilt by PROC_ARCANA_Aspect_Init on a new game and on every load):
//   DB_ARCANA_Aspect(Aspect, ProgressStatus, AwakenedStatus, TierIPassive)
//   DB_ARCANA_AspectTier(Aspect, Tier, Passive, TierStatus, Goal)   gen_awakening_progress.pl reads these rows
//   DB_ARCANA_AspectShow(Aspect, Mode)                              how the countdown is shown
//   DB_ARCANA_AspectProgress(Aspect, Tier, Status)                  the countdown status of each tier
// DBs (state; the reset deletes every row of the character):
//   DB_ARCANA_Aspect_InCombat(Char, Combat)   the roster of every fight. Deleted on LeftCombat, or when the fight
//                                             ends with the character still listed. Read by most Aspect goals.
//   DB_ARCANA_Aspect_Steps(Char, N)           the plain count of PROC_ARCANA_Aspect_Step.
//   DB_ARCANA_AspectReached(Char, Tier)       tiers awakened this fight. Keeps no Aspect: a goal that reads it
//                                             guards with the subclass's own passive or status.
//   DB_ARCANA_AspectNext(Char, Tier)          the next tier the level unlocks, for the countdown.
// The countdown block (Display / ShowLeft / ShowNumber) is generated by gen_awakening_progress.pl; change it
// there.

//REGION Tables
// Cleared before it is filled: an old save keeps the rows with old values.
PROC
PROC_ARCANA_Aspect_Init()
AND
DB_ARCANA_Aspect(_Aspect, _Progress, _Awakened, _TierI)
THEN
NOT DB_ARCANA_Aspect(_Aspect, _Progress, _Awakened, _TierI);

PROC
PROC_ARCANA_Aspect_Init()
AND
DB_ARCANA_AspectTier(_Aspect, _Tier, _Passive, _Status, _Goal)
THEN
NOT DB_ARCANA_AspectTier(_Aspect, _Tier, _Passive, _Status, _Goal);

PROC
PROC_ARCANA_Aspect_Init()
AND
DB_ARCANA_AspectShow(_Aspect, _Mode)
THEN
NOT DB_ARCANA_AspectShow(_Aspect, _Mode);

PROC
PROC_ARCANA_Aspect_Init()
THEN
// Aspect, progress status, technical status (Tier I reached), Tier I passive
DB_ARCANA_Aspect("FOOL", "ASPECT_PROGRESS_FOOL", "ASPECT_AWAKENED_FOOL", "Arcana_Passive_Aspect_Fool_I");
DB_ARCANA_Aspect("EMPRESS", "ASPECT_PROGRESS_EMPRESS", "ASPECT_AWAKENED_EMPRESS", "Arcana_Passive_Aspect_Empress_I");
DB_ARCANA_Aspect("STAR", "ASPECT_PROGRESS_STAR", "ASPECT_AWAKENED_STAR", "Arcana_Passive_Aspect_Star_I");
DB_ARCANA_Aspect("DEATH", "ASPECT_PROGRESS_DEATH", "ASPECT_AWAKENED_DEATH", "Arcana_Passive_Aspect_Death_I");
DB_ARCANA_Aspect("TEMPERANCE", "ASPECT_PROGRESS_TEMPERANCE", "ASPECT_AWAKENED_TEMPERANCE", "Arcana_Passive_Aspect_Temperance_I");
DB_ARCANA_Aspect("WHEEL", "ASPECT_PROGRESS_WHEEL", "ASPECT_AWAKENED_WHEEL", "Arcana_Passive_Aspect_WheelOfFortune_I");
// Aspect, tier, the passive that unlocks it by level, the tier's status, the goal in the subclass's unit
DB_ARCANA_AspectTier("FOOL", 1, "Arcana_Passive_Aspect_Fool_I", "ASPECT_AWAKENED_FOOL_I", 3);
DB_ARCANA_AspectTier("FOOL", 2, "Arcana_Passive_Aspect_Fool_II", "ASPECT_AWAKENED_FOOL_II", 5);

DB_ARCANA_AspectTier("EMPRESS", 1, "Arcana_Passive_Aspect_Empress_I", "ASPECT_AWAKENED_EMPRESS_I", 4);
DB_ARCANA_AspectTier("EMPRESS", 2, "Arcana_Passive_Aspect_Empress_II", "ASPECT_AWAKENED_EMPRESS_II", 7);
DB_ARCANA_AspectTier("EMPRESS", 3, "Arcana_Passive_Aspect_Empress_III", "ASPECT_AWAKENED_EMPRESS_III", 10);
DB_ARCANA_AspectTier("STAR", 1, "Arcana_Passive_Aspect_Star_I", "ASPECT_AWAKENED_STAR_I", 500);
DB_ARCANA_AspectTier("STAR", 2, "Arcana_Passive_Aspect_Star_II", "ASPECT_AWAKENED_STAR_II", 800);
DB_ARCANA_AspectTier("STAR", 3, "Arcana_Passive_Aspect_Star_III", "ASPECT_AWAKENED_STAR_III", 1200);
DB_ARCANA_AspectTier("DEATH", 1, "Arcana_Passive_Aspect_Death_I", "ASPECT_AWAKENED_DEATH_I", 300);
DB_ARCANA_AspectTier("DEATH", 2, "Arcana_Passive_Aspect_Death_II", "ASPECT_AWAKENED_DEATH_II", 500);
DB_ARCANA_AspectTier("DEATH", 3, "Arcana_Passive_Aspect_Death_III", "ASPECT_AWAKENED_DEATH_III", 700);
DB_ARCANA_AspectTier("TEMPERANCE", 1, "Arcana_Passive_Aspect_Temperance_I", "ASPECT_AWAKENED_TEMPERANCE_I", 3);
DB_ARCANA_AspectTier("TEMPERANCE", 2, "Arcana_Passive_Aspect_Temperance_II", "ASPECT_AWAKENED_TEMPERANCE_II", 6);
DB_ARCANA_AspectTier("TEMPERANCE", 3, "Arcana_Passive_Aspect_Temperance_III", "ASPECT_AWAKENED_TEMPERANCE_III", 15);
DB_ARCANA_AspectTier("WHEEL", 1, "Arcana_Passive_Aspect_WheelOfFortune_I", "ASPECT_AWAKENED_WHEEL_I", 3);
DB_ARCANA_AspectTier("WHEEL", 2, "Arcana_Passive_Aspect_WheelOfFortune_II", "ASPECT_AWAKENED_WHEEL_II", 5);
DB_ARCANA_AspectTier("WHEEL", 3, "Arcana_Passive_Aspect_WheelOfFortune_III", "ASPECT_AWAKENED_WHEEL_III", 8);
// how the progress is shown: COUNT = the number itself; HEAL / LOSS = converted back to hit points
DB_ARCANA_AspectShow("FOOL", "COUNT");
DB_ARCANA_AspectShow("EMPRESS", "COUNT");
DB_ARCANA_AspectShow("STAR", "HEAL");
DB_ARCANA_AspectShow("DEATH", "LOSS");
DB_ARCANA_AspectShow("TEMPERANCE", "COUNT");
DB_ARCANA_AspectShow("WHEEL", "COUNT");
// INITSECTION does not run again in an existing save.
IF
SavegameLoaded()
THEN
PROC_ARCANA_Aspect_Init();
//END_REGION

//REGION The roster of the fight
IF
EnteredCombat((CHARACTER)_Char, _Combat)
THEN
DB_ARCANA_Aspect_InCombat(_Char, _Combat);

IF
LeftCombat((CHARACTER)_Char, _Combat)
AND
DB_ARCANA_Aspect_InCombat(_Char, _Combat)
THEN
NOT DB_ARCANA_Aspect_InCombat(_Char, _Combat);
PROC_ARCANA_Aspect_Reset(_Char);

// A character still listed when its fight ends (no LeftCombat came for it, as after a death) is reset now, so no
// Aspect status or row outlives the fight. One that already left has no row here: nothing runs twice.
IF
CombatEnded(_Combat)
AND
DB_ARCANA_Aspect_InCombat(_Char, _Combat)
THEN
NOT DB_ARCANA_Aspect_InCombat(_Char, _Combat);
PROC_ARCANA_Aspect_Reset(_Char);
//END_REGION

//REGION Plain count (High Priestess, Windborn, Kingmaker, Resonant): +1 per event, in combat only
PROC
PROC_ARCANA_Aspect_Step((CHARACTER)_Char, (STRING)_Aspect)
AND
IsInCombat(_Char, 1)
AND
NOT DB_ARCANA_Aspect_Steps(_Char, _)
THEN
DB_ARCANA_Aspect_Steps(_Char, 0);

PROC
PROC_ARCANA_Aspect_Step((CHARACTER)_Char, (STRING)_Aspect)
AND
IsInCombat(_Char, 1)
AND
DB_ARCANA_Aspect_Steps(_Char, _Old)
AND
IntegerSum(_Old, 1, _New)
THEN
NOT DB_ARCANA_Aspect_Steps(_Char, _Old);
DB_ARCANA_Aspect_Steps(_Char, _New);
PROC_ARCANA_Aspect_Progress(_Char, _Aspect, _New);
//END_REGION

//REGION Tiers
// the same value is offered to the three goals; each tier reached awakens
PROC
PROC_ARCANA_Aspect_Progress((CHARACTER)_Char, (STRING)_Aspect, (INTEGER)_Value)
THEN
PROC_ARCANA_Aspect_Offer(_Char, _Aspect, 1, _Value);
PROC_ARCANA_Aspect_Offer(_Char, _Aspect, 2, _Value);
PROC_ARCANA_Aspect_Offer(_Char, _Aspect, 3, _Value);
PROC_ARCANA_Aspect_FindNext(_Char, _Aspect);
PROC_ARCANA_Aspect_Display(_Char, _Aspect, _Value);

PROC
PROC_ARCANA_Aspect_Offer((CHARACTER)_Char, (STRING)_Aspect, (INTEGER)_Tier, (INTEGER)_Value)
AND
DB_ARCANA_AspectTier(_Aspect, _Tier, _Passive, _Status, _Goal)
AND
_Value >= _Goal
AND
HasPassive(_Char, _Passive, 1)
AND
IsInCombat(_Char, 1)
AND
NOT DB_ARCANA_AspectReached(_Char, _Tier)
THEN
DB_ARCANA_AspectReached(_Char, _Tier);
PROC_ARCANA_Aspect_Awaken(_Char, _Aspect, _Tier, _Status);

// Tier I also awakens the Aspect (the card, the VFX and the Tier I effects); the two rules run in this order
PROC
PROC_ARCANA_Aspect_Awaken((CHARACTER)_Char, (STRING)_Aspect, (INTEGER)_Tier, (STRING)_Status)
AND
_Tier == 1
AND
DB_ARCANA_Aspect(_Aspect, _Progress, _Awakened, _TierI)
THEN
ApplyStatus(_Char, _Awakened, -1.0, 1, _Char);
PROC_ARCANA_Aspect_OnAwakened(_Char, _Aspect);

// the tier's status replaces the one below (same StackId, higher priority) and carries the bonuses of all of them
PROC
PROC_ARCANA_Aspect_Awaken((CHARACTER)_Char, (STRING)_Aspect, (INTEGER)_Tier, (STRING)_Status)
THEN
ApplyStatus(_Char, _Status, -1.0, 1, _Char);
PROC_ARCANA_Aspect_OnTier(_Char, _Aspect, _Tier);
//END_REGION

//REGION Next tier and the countdown on screen
// the lowest tier the level unlocks that has not awakened yet; with none, the countdown goes away
PROC
PROC_ARCANA_Aspect_FindNext((CHARACTER)_Char, (STRING)_Aspect)
AND
DB_ARCANA_AspectNext(_Char, _Old)
THEN
NOT DB_ARCANA_AspectNext(_Char, _Old);

PROC
PROC_ARCANA_Aspect_FindNext((CHARACTER)_Char, (STRING)_Aspect)
THEN
PROC_ARCANA_Aspect_TryNext(_Char, _Aspect, 1);
PROC_ARCANA_Aspect_TryNext(_Char, _Aspect, 2);
PROC_ARCANA_Aspect_TryNext(_Char, _Aspect, 3);

PROC
PROC_ARCANA_Aspect_TryNext((CHARACTER)_Char, (STRING)_Aspect, (INTEGER)_Tier)
AND
NOT DB_ARCANA_AspectNext(_Char, _)
AND
NOT DB_ARCANA_AspectReached(_Char, _Tier)
AND
DB_ARCANA_AspectTier(_Aspect, _Tier, _Passive, _, _)
AND
HasPassive(_Char, _Passive, 1)
THEN
DB_ARCANA_AspectNext(_Char, _Tier);

// Generated by gen_awakening_progress.pl: the number counts DOWN -- what is left to the next tier, in the
// Aspect's unit (COUNT: the goal's own unit; HEAL: hit points to heal; LOSS: hit points to lose). One status per
// tier (DB_ARCANA_AspectProgress), so its text names the tier it awakens. Its table is part of the Init.
PROC
PROC_ARCANA_Aspect_Init()
AND
DB_ARCANA_AspectProgress(_Aspect, _Tier, _Status)
THEN
NOT DB_ARCANA_AspectProgress(_Aspect, _Tier, _Status);

DB_ARCANA_AspectProgress("FOOL", 3, "ASPECT_PROGRESS_FOOL_III");
DB_ARCANA_AspectProgress("EMPRESS", 1, "ASPECT_PROGRESS_EMPRESS_I");
DB_ARCANA_AspectProgress("EMPRESS", 2, "ASPECT_PROGRESS_EMPRESS_II");
DB_ARCANA_AspectProgress("EMPRESS", 3, "ASPECT_PROGRESS_EMPRESS_III");
DB_ARCANA_AspectProgress("TEMPERANCE", 1, "ASPECT_PROGRESS_TEMPERANCE_I");
DB_ARCANA_AspectProgress("TEMPERANCE", 2, "ASPECT_PROGRESS_TEMPERANCE_II");
DB_ARCANA_AspectProgress("TEMPERANCE", 3, "ASPECT_PROGRESS_TEMPERANCE_III");

DB_ARCANA_AspectProgress("WHEEL", 2, "ASPECT_PROGRESS_WHEEL_II");
DB_ARCANA_AspectProgress("WHEEL", 3, "ASPECT_PROGRESS_WHEEL_III");

DB_ARCANA_AspectProgress("STAR", 3, "ASPECT_PROGRESS_STAR_III");
DB_ARCANA_AspectProgress("DEATH", 1, "ASPECT_PROGRESS_DEATH_I");
DB_ARCANA_AspectProgress("DEATH", 2, "ASPECT_PROGRESS_DEATH_II");
DB_ARCANA_AspectProgress("DEATH", 3, "ASPECT_PROGRESS_DEATH_III");

// no next tier (Tier III reached, or the level unlocks none): it goes away
PROC
PROC_ARCANA_Aspect_Display((CHARACTER)_Char, (STRING)_Aspect, (INTEGER)_Value)
AND
NOT DB_ARCANA_AspectNext(_Char, _)
THEN
PROC_ARCANA_Aspect_HideProgress(_Char, _Aspect);

PROC
PROC_ARCANA_Aspect_Display((CHARACTER)_Char, (STRING)_Aspect, (INTEGER)_Value)
AND
DB_ARCANA_AspectNext(_Char, _Tier)
AND
DB_ARCANA_AspectTier(_Aspect, _Tier, _, _, _Goal)
AND
IntegerSubtract(_Goal, _Value, _Left)
THEN
PROC_ARCANA_Aspect_ShowLeft(_Char, _Aspect, _Tier, _Left);

PROC
PROC_ARCANA_Aspect_ShowLeft((CHARACTER)_Char, (STRING)_Aspect, (INTEGER)_Tier, (INTEGER)_Left)
AND
DB_ARCANA_AspectShow(_Aspect, "COUNT")
THEN
PROC_ARCANA_Aspect_ShowNumber(_Char, _Aspect, _Tier, _Left);

// Star: the value is healing x 100 / level -- back to hit points, rounded up
PROC
PROC_ARCANA_Aspect_ShowLeft((CHARACTER)_Char, (STRING)_Aspect, (INTEGER)_Tier, (INTEGER)_Left)
AND
DB_ARCANA_AspectShow(_Aspect, "HEAL")
AND
GetLevel(_Char, _Level)
AND
IntegerProduct(_Left, _Level, _X)
AND
IntegerSum(_X, 99, _Up)
AND
IntegerDivide(_Up, 100, _HP)
THEN
PROC_ARCANA_Aspect_ShowNumber(_Char, _Aspect, _Tier, _HP);

// Death: the value is loss x 1000 / max HP -- back to hit points, rounded up
PROC
PROC_ARCANA_Aspect_ShowLeft((CHARACTER)_Char, (STRING)_Aspect, (INTEGER)_Tier, (INTEGER)_Left)
AND
DB_ARCANA_AspectShow(_Aspect, "LOSS")
AND
GetMaxHitpoints(_Char, _Max)
AND
IntegerProduct(_Left, _Max, _X)
AND
IntegerSum(_X, 999, _Up)
AND
IntegerDivide(_Up, 1000, _HP)
THEN
PROC_ARCANA_Aspect_ShowNumber(_Char, _Aspect, _Tier, _HP);

// the number is the frozen duration (6 s per turn); 0 or less: nothing to show
PROC
PROC_ARCANA_Aspect_ShowNumber((CHARACTER)_Char, (STRING)_Aspect, (INTEGER)_Tier, (INTEGER)_N)
AND
_N >= 1
AND
DB_ARCANA_AspectProgress(_Aspect, _Tier, _Status)
AND
IntegerProduct(_N, 6, _Secs)
AND
IntegerToReal(_Secs, _Dur)
THEN
ApplyStatus(_Char, _Status, _Dur, 1, _Char);

PROC
PROC_ARCANA_Aspect_ShowNumber((CHARACTER)_Char, (STRING)_Aspect, (INTEGER)_Tier, (INTEGER)_N)
AND
_N < 1
THEN
PROC_ARCANA_Aspect_HideProgress(_Char, _Aspect);

PROC
PROC_ARCANA_Aspect_HideProgress((CHARACTER)_Char, (STRING)_Aspect)
AND
DB_ARCANA_AspectProgress(_Aspect, _Tier, _Status)
THEN
RemoveStatus(_Char, _Status, _Char);

PROC
PROC_ARCANA_Aspect_Reset((CHARACTER)_Char)
AND
DB_ARCANA_AspectProgress(_Aspect, _Tier, _Status)
THEN
RemoveStatus(_Char, _Status, _Char);
//END_REGION

//REGION Cleanup when the fight ends (each Aspect's goal adds its own reset rules)
PROC
PROC_ARCANA_Aspect_Reset((CHARACTER)_Char)
AND
DB_ARCANA_Aspect_Steps(_Char, _N)
THEN
NOT DB_ARCANA_Aspect_Steps(_Char, _N);

PROC
PROC_ARCANA_Aspect_Reset((CHARACTER)_Char)
AND
DB_ARCANA_AspectReached(_Char, _Tier)
THEN
NOT DB_ARCANA_AspectReached(_Char, _Tier);

PROC
PROC_ARCANA_Aspect_Reset((CHARACTER)_Char)
AND
DB_ARCANA_AspectNext(_Char, _Tier)
THEN
NOT DB_ARCANA_AspectNext(_Char, _Tier);

PROC
PROC_ARCANA_Aspect_Reset((CHARACTER)_Char)
AND
DB_ARCANA_Aspect(_Aspect, _Progress, _Awakened, _TierI)
THEN
RemoveStatus(_Char, _Progress, _Char);
RemoveStatus(_Char, _Awakened, _Char);

PROC
PROC_ARCANA_Aspect_Reset((CHARACTER)_Char)
AND
DB_ARCANA_AspectTier(_Aspect, _Tier, _Passive, _Status, _Goal)
THEN
RemoveStatus(_Char, _Status, _Char);
//END_REGION
