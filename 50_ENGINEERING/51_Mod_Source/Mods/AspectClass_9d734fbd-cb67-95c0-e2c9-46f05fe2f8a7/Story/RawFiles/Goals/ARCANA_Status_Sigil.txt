KBSECTION
// A marca fica em nome de quem a aplicou: a SIGIL so detona com dano de quem a aplicou ou de uma
// invocacao dele (HasStatus('SIGIL', context.Target, context.Source) nas RemoveConditions).
// O clone marca em nome do Deceiver dono dele, para que os dois possam detonar as marcas um do outro:
// o Deceiver pela origem da marca, o clone pelo GetSummoner.
IF
StatusApplied(_Target, "SIGIL_TECH", _Caster, _)
AND
DB_ARCANA_MirrorImageOf((CHARACTER)_Caster, _Owner)
AND
HasActiveStatus(_Target, "SIGIL", 0)
THEN
ApplyStatus(_Target, "SIGIL", -6.0, 1, _Owner);

IF
StatusApplied(_Target, "SIGIL_TECH", _Caster, _)
AND
NOT DB_ARCANA_MirrorImageOf((CHARACTER)_Caster, _)
AND
HasActiveStatus(_Target, "SIGIL", 0)
THEN
ApplyStatus(_Target, "SIGIL", -6.0, 1, _Caster);

//REGION Task 5: a detonation adds Doubt (The Veil)
// SIGIL.OnRemoveFunctors applies SIGIL_DETONATED_TECH when the mark is triggered, but a status applied from an
// OnRemoveFunctors has no guaranteed source, so the owner of each mark is remembered here when the mark lands (the two
// rules above always put it in the Deceiver's name, also when a Double lays it). On the detonation the foe gets ONE
// DOUBT_ADD_TECH in her name: the Doubt goal (ARCANA_Deceiver_Doubt.txt) raises the stack and credits her.
// Own DB on purpose: ARCANA_Aspect_Empress.txt keeps DB_ARCANA_SigilOwner for the same event and deletes its row
// there; two goals reacting to one event in an unknown order must not share a row.
//
// DB_ARCANA_SigilDoubtOwner(Foe, Owner)  the Deceiver whose mark the foe carries (the latest one applied). A row can
//                                        outlive a mark removed without a detonation; it is harmless (only a new
//                                        mark, which replaces it, can detonate) and goes on the foe's death or when
//                                        either of them leaves combat.
IF
StatusApplied((CHARACTER)_Foe, "SIGIL", (CHARACTER)_Owner, _)
THEN
PROC_ARCANA_Sigil_SetDoubtOwner(_Foe, _Owner);

PROC
PROC_ARCANA_Sigil_SetDoubtOwner((CHARACTER)_Foe, (CHARACTER)_Owner)
AND
DB_ARCANA_SigilDoubtOwner(_Foe, _Old)
THEN
NOT DB_ARCANA_SigilDoubtOwner(_Foe, _Old);

PROC
PROC_ARCANA_Sigil_SetDoubtOwner((CHARACTER)_Foe, (CHARACTER)_Owner)
THEN
DB_ARCANA_SigilDoubtOwner(_Foe, _Owner);

IF
StatusApplied((CHARACTER)_Foe, "SIGIL_DETONATED_TECH", _, _)
AND
DB_ARCANA_SigilDoubtOwner(_Foe, _Owner)
THEN
NOT DB_ARCANA_SigilDoubtOwner(_Foe, _Owner);
PROC_ARCANA_Sigil_Doubt(_Foe, _Owner);

// A detonation that kills (the mark's damage is skipped on a killing blow, the detonation status is not) adds nothing.
PROC
PROC_ARCANA_Sigil_Doubt((CHARACTER)_Foe, (CHARACTER)_Owner)
AND
IsDead(_Foe, 0)
THEN
ApplyStatus(_Foe, "DOUBT_ADD_TECH", 0.0, 1, _Owner);

IF
Died((CHARACTER)_Foe)
AND
DB_ARCANA_SigilDoubtOwner(_Foe, _Owner)
THEN
NOT DB_ARCANA_SigilDoubtOwner(_Foe, _Owner);

// PROC_ARCANA_Aspect_Reset (ARCANA_Aspect_Common.txt) runs when a character leaves combat: drop its rows either way.
PROC
PROC_ARCANA_Aspect_Reset((CHARACTER)_Char)
AND
DB_ARCANA_SigilDoubtOwner(_Foe, _Char)
THEN
NOT DB_ARCANA_SigilDoubtOwner(_Foe, _Char);

PROC
PROC_ARCANA_Aspect_Reset((CHARACTER)_Char)
AND
DB_ARCANA_SigilDoubtOwner(_Char, _Owner)
THEN
NOT DB_ARCANA_SigilDoubtOwner(_Char, _Owner);
//END_REGION
