// Overload (Resonant): the meter that fills as her emotions land (+1 per peak, +2 per reaction, R12), sized by
// level (25: FEAT_Resonant_Overload) and shown as a frozen duration of N x 6 s, like the Awakening countdown.
// Full -> Overloaded for turns equal to the level (R29). It carries over between fights, settling to half.
// Generated by gen_emotion_alchemy.pl from the wheel table, so the status lists never drift: edit the
// generator, not this file.
//
// DBs (tables, rebuilt by PROC_ARCANA_Resonant_Overload_Init on a new game and on every load):
//   DB_ARCANA_Resonant_OverloadPoints(Status, Points)   DB_ARCANA_Resonant_OverloadMax(Level, Max)
//   DB_ARCANA_Resonant_Reaction(Status)   DB_ARCANA_Resonant_Landed(Status)   DB_ARCANA_Resonant_RiderBand(Level, Tech)
// DBs (state):
//   DB_ARCANA_Resonant_Overload(Resonant, N)    the count. Deleted when it fills, resets or settles to 0.
//   DB_ARCANA_Resonant_OverloadFull(Resonant)   Overloaded. Deleted by the reset or the settle. Read by
//                                               ARCANA_Aspect_Temperance.txt.
//   DB_ARCANA_Resonant_Settling(Resonant)       the settle is removing Overloaded: that StatusRemoved must not
//                                               reset the meter.
//   DB_ARCANA_Resonant_RiderUsed(Resonant)      the once-per-turn rider was used. Deleted at her turn start.
// Entry point: PROC_ARCANA_Resonant_Overload_Fill (Tier III of ARCANA_Aspect_Temperance.txt).

//REGION Refund (R16): a reaction gives the Surge's slot back; RESONANT_REFUND_TECH holds the once-per-turn guard
IF
StatusApplied(_Target, _Status, (CHARACTER)_Resonant, _)
AND
DB_ARCANA_Resonant_Reaction(_Status)
AND
HasPassive(_Resonant, "Arcana_Passive_EmotionalAlchemy", 1)
THEN
ApplyStatus(_Resonant, "RESONANT_REFUND_TECH", 0.0, 1, _Resonant);
//END_REGION

//REGION Overload count: +1 per peak that lands, +2 per reaction, at any time; it carries over between combats (R12, R31)
IF
StatusApplied(_Target, _Status, (CHARACTER)_Resonant, _)
AND
DB_ARCANA_Resonant_OverloadPoints(_Status, _Pts)
AND
HasPassive(_Resonant, "Arcana_Passive_Overload", 1)
AND
NOT DB_ARCANA_Resonant_OverloadFull(_Resonant)
THEN
PROC_ARCANA_Resonant_Overload_Add(_Resonant, _Pts);

PROC
PROC_ARCANA_Resonant_Overload_Add((CHARACTER)_Resonant, (INTEGER)_Pts)
AND
NOT DB_ARCANA_Resonant_Overload(_Resonant, _)
THEN
DB_ARCANA_Resonant_Overload(_Resonant, 0);

PROC
PROC_ARCANA_Resonant_Overload_Add((CHARACTER)_Resonant, (INTEGER)_Pts)
AND
DB_ARCANA_Resonant_Overload(_Resonant, _Old)
AND
IntegerSum(_Old, _Pts, _New)
AND
GetLevel(_Resonant, _Lvl)
AND
DB_ARCANA_Resonant_OverloadMax(_Lvl, _Max)
THEN
NOT DB_ARCANA_Resonant_Overload(_Resonant, _Old);
PROC_ARCANA_Resonant_Overload_Set(_Resonant, _New, _Max, _Lvl);

PROC
PROC_ARCANA_Resonant_Overload_Set((CHARACTER)_Resonant, (INTEGER)_New, (INTEGER)_Max, (INTEGER)_Lvl)
AND
_New < _Max
THEN
DB_ARCANA_Resonant_Overload(_Resonant, _New);
PROC_ARCANA_Resonant_Overload_Show(_Resonant, _New, _Max);

// R29: full -> Overloaded for turns equal to the character level (ApplyStatus duration is in seconds, 6 per turn)
PROC
PROC_ARCANA_Resonant_Overload_Set((CHARACTER)_Resonant, (INTEGER)_New, (INTEGER)_Max, (INTEGER)_Lvl)
AND
_New >= _Max
AND
IntegerProduct(_Lvl, 6, _Secs)
AND
IntegerToReal(_Secs, _Dur)
THEN
DB_ARCANA_Resonant_Overload(_Resonant, _Max);
DB_ARCANA_Resonant_OverloadFull(_Resonant);
RemoveStatus(_Resonant, "RESONANT_OVERLOAD", _Resonant);
ApplyStatus(_Resonant, "RESONANT_OVERLOAD_FULL", _Dur, 1, _Resonant);

// R31: the meter shows the percentage of its maximum (1-99), as the PERCENT mode of ARCANA_Aspect_Common
PROC
PROC_ARCANA_Resonant_Overload_Show((CHARACTER)_Resonant, (INTEGER)_N, (INTEGER)_Max)
AND
IntegerProduct(_N, 100, _N100)
AND
IntegerDivide(_N100, _Max, _Pct)
AND
_Pct < 1
THEN
PROC_ARCANA_Resonant_Overload_ShowPct(_Resonant, 1);

PROC
PROC_ARCANA_Resonant_Overload_Show((CHARACTER)_Resonant, (INTEGER)_N, (INTEGER)_Max)
AND
IntegerProduct(_N, 100, _N100)
AND
IntegerDivide(_N100, _Max, _Pct)
AND
_Pct >= 1
AND
_Pct < 100
THEN
PROC_ARCANA_Resonant_Overload_ShowPct(_Resonant, _Pct);

PROC
PROC_ARCANA_Resonant_Overload_ShowPct((CHARACTER)_Resonant, (INTEGER)_P)
AND
IntegerProduct(_P, 6, _Secs)
AND
IntegerToReal(_Secs, _Dur)
THEN
ApplyStatus(_Resonant, "RESONANT_OVERLOAD", _Dur, 1, _Resonant);
//END_REGION

//REGION Overload reset: Emotional Overload released, or Overloaded ran its course
PROC
PROC_ARCANA_Resonant_Overload_Reset((CHARACTER)_Resonant)
AND
DB_ARCANA_Resonant_Overload(_Resonant, _N)
THEN
NOT DB_ARCANA_Resonant_Overload(_Resonant, _N);

PROC
PROC_ARCANA_Resonant_Overload_Reset((CHARACTER)_Resonant)
AND
DB_ARCANA_Resonant_OverloadFull(_Resonant)
THEN
NOT DB_ARCANA_Resonant_OverloadFull(_Resonant);

// R32: the meter stays on the character, empty
PROC
PROC_ARCANA_Resonant_Overload_Reset((CHARACTER)_Resonant)
THEN
RemoveStatus(_Resonant, "RESONANT_OVERLOAD_FULL", _Resonant);
PROC_ARCANA_Resonant_Overload_ShowEmpty(_Resonant);

IF
StatusApplied((CHARACTER)_Resonant, "RESONANT_OVERLOAD_SPENT_TECH", _, _)
THEN
PROC_ARCANA_Resonant_Overload_Reset(_Resonant);

// R29: Overloaded ran out (or was removed) -- the Resonant has steadied, the meter is empty
IF
StatusRemoved((CHARACTER)_Resonant, "RESONANT_OVERLOAD_FULL", _, _)
AND
NOT DB_ARCANA_Resonant_Settling(_Resonant)
THEN
PROC_ARCANA_Resonant_Overload_Reset(_Resonant);

IF
StatusRemoved((CHARACTER)_Resonant, "RESONANT_OVERLOAD_FULL", _, _)
AND
DB_ARCANA_Resonant_Settling(_Resonant)
THEN
NOT DB_ARCANA_Resonant_Settling(_Resonant);
//END_REGION

//REGION The empty meter (R32): always shown, at 0% too -- a permanent status with no number
PROC
PROC_ARCANA_Resonant_Overload_ShowEmpty((CHARACTER)_Resonant)
AND
HasPassive(_Resonant, "Arcana_Passive_Overload", 1)
AND
NOT DB_ARCANA_Resonant_Overload(_Resonant, _)
AND
NOT DB_ARCANA_Resonant_OverloadFull(_Resonant)
THEN
ApplyStatus(_Resonant, "RESONANT_OVERLOAD", -1.0, 1, _Resonant);

IF
SavegameLoaded()
AND
DB_Players(_Resonant)
THEN
PROC_ARCANA_Resonant_Overload_ShowEmpty(_Resonant);

// the passive arrives with the class (level 1): a new Resonant, a new party member, or the first fight
IF
LeveledUp(_Resonant)
THEN
PROC_ARCANA_Resonant_Overload_ShowEmpty(_Resonant);

IF
CharacterJoinedParty(_Resonant)
THEN
PROC_ARCANA_Resonant_Overload_ShowEmpty(_Resonant);

IF
EnteredCombat((CHARACTER)_Resonant, _)
THEN
PROC_ARCANA_Resonant_Overload_ShowEmpty(_Resonant);
//END_REGION

//REGION R51: once per turn, a Surge that lands on an enemy while Overloaded deals it the Overloaded damage
IF
StatusApplied((CHARACTER)_Target, _Status, (CHARACTER)_Resonant, _)
AND
DB_ARCANA_Resonant_Landed(_Status)
AND
HasActiveStatus(_Resonant, "RESONANT_OVERLOAD_FULL", 1)
AND
IsEnemy(_Resonant, _Target, 1)
AND
NOT DB_ARCANA_Resonant_RiderUsed(_Resonant)
AND
GetLevel(_Resonant, _Lvl)
AND
DB_ARCANA_Resonant_RiderBand(_Lvl, _Tech)
THEN
DB_ARCANA_Resonant_RiderUsed(_Resonant);
ApplyStatus(_Target, _Tech, 0.0, 1, _Resonant);

IF
TurnStarted((CHARACTER)_Resonant)
AND
DB_ARCANA_Resonant_RiderUsed(_Resonant)
THEN
NOT DB_ARCANA_Resonant_RiderUsed(_Resonant);
//END_REGION

//REGION R53: when a fight ends the meter settles to half and Overloaded ends (not-full first: the full case sets the count)
IF
LeftCombat((CHARACTER)_Resonant, _)
AND
HasPassive(_Resonant, "Arcana_Passive_Overload", 1)
THEN
PROC_ARCANA_Resonant_Overload_Settle(_Resonant);

PROC
PROC_ARCANA_Resonant_Overload_Settle((CHARACTER)_Resonant)
AND
NOT DB_ARCANA_Resonant_OverloadFull(_Resonant)
AND
DB_ARCANA_Resonant_Overload(_Resonant, _N)
AND
IntegerDivide(_N, 2, _Half)
THEN
NOT DB_ARCANA_Resonant_Overload(_Resonant, _N);
PROC_ARCANA_Resonant_Overload_Keep(_Resonant, _Half);

PROC
PROC_ARCANA_Resonant_Overload_Settle((CHARACTER)_Resonant)
AND
DB_ARCANA_Resonant_OverloadFull(_Resonant)
AND
DB_ARCANA_Resonant_Overload(_Resonant, _N)
AND
IntegerDivide(_N, 2, _Half)
THEN
DB_ARCANA_Resonant_Settling(_Resonant);
NOT DB_ARCANA_Resonant_OverloadFull(_Resonant);
NOT DB_ARCANA_Resonant_Overload(_Resonant, _N);
RemoveStatus(_Resonant, "RESONANT_OVERLOAD_FULL", _Resonant);
PROC_ARCANA_Resonant_Overload_Keep(_Resonant, _Half);

PROC
PROC_ARCANA_Resonant_Overload_Keep((CHARACTER)_Resonant, (INTEGER)_Half)
AND
_Half > 0
AND
GetLevel(_Resonant, _Lvl)
AND
DB_ARCANA_Resonant_OverloadMax(_Lvl, _Max)
THEN
DB_ARCANA_Resonant_Overload(_Resonant, _Half);
PROC_ARCANA_Resonant_Overload_Show(_Resonant, _Half, _Max);

PROC
PROC_ARCANA_Resonant_Overload_Keep((CHARACTER)_Resonant, (INTEGER)_Half)
AND
_Half == 0
THEN
PROC_ARCANA_Resonant_Overload_ShowEmpty(_Resonant);
//END_REGION

//REGION R56: waking Tier III fills the meter (ARCANA_Aspect_Temperance calls this)
PROC
PROC_ARCANA_Resonant_Overload_Fill((CHARACTER)_Resonant)
AND
GetLevel(_Resonant, _Lvl)
AND
DB_ARCANA_Resonant_OverloadMax(_Lvl, _Max)
THEN
PROC_ARCANA_Resonant_Overload_Add(_Resonant, _Max);
//END_REGION

//REGION R51: Emotional Overload leaves vanilla Mental Fatigue for the Resonant's level (its technical status caps it at 7)
IF
StatusApplied((CHARACTER)_Target, "RESONANT_FATIGUE_TECH", (CHARACTER)_Resonant, _)
AND
GetLevel(_Resonant, _Lvl)
AND
IntegerProduct(_Lvl, 6, _Secs)
AND
IntegerToReal(_Secs, _Dur)
THEN
ApplyStatus(_Target, "MAG_PSYCHIC_MENTAL_FATIGUE", _Dur, 1, _Resonant);
ApplyStatus(_Target, "MAG_PSYCHIC_MENTAL_FATIGUE_DURATION_TECHNICAL", 6.0, 1, _Resonant);
//END_REGION
