// Aspect of The High Priestess (the Deceiver; internal ids keep "Empress", spec D12).
//   Goal: every stack of Doubt she applies is one step (DB_ARCANA_DoubtStep rows, written by
//   ARCANA_Deceiver_Doubt.txt); 4 / 7 / 10 in ARCANA_Aspect_Common.txt.
//   Tier I (Veiled) lives in the DOUBT statuses. Tier II (Two Pillars): once per turn, when an enemy becomes
//   Deluded in her name, the nearest unmarked enemy within 6 m gains her Sigil. Tier III: Ready: Which One? (the
//   swap is here). Aspect card: The Veil Lifts (every Double shatters; survivors with Doubt become Deluded).
//
// DBs:
//   DB_ARCANA_EmpressChained(Deceiver)            Two Pillars already chained this turn. Deleted at her turn start
//                                                 and by the reset.
//   DB_ARCANA_EmpressBest(Deceiver, Foe, Dist)    transient: the nearest candidate for the Sigil.
//   DB_ARCANA_WhichOneBest(Owner, Double, Dist)   transient: the nearest Double for Which One?. Also cleared by the
//                                                 reset.
//   DB_ARCANA_VeilLiftsBy(Foe, Owner)             an enemy The Veil Lifts touched, waiting 900 ms for its Doubt.
//                                                 Deleted by the timer and by the reset.
// Reads: DB_ARCANA_DoubtStep (consumes it), DB_ARCANA_DoubtCount (ARCANA_Deceiver_Doubt.txt); DB_ARCANA_Double,
// DB_ARCANA_DoubleGone (ARCANA_Deceiver_Doubles.txt); DB_ARCANA_Aspect_InCombat, DB_ARCANA_AspectReached
// (ARCANA_Aspect_Common.txt); DB_ARCANA_CardFamily. Calls PROC_ARCANA_Doubt_Raise, PROC_ARCANA_Doubles_Retire.
// Hooks: PROC_ARCANA_Aspect_Reset.

//REGION Goal: Doubt applied
IF
DB_ARCANA_DoubtStep(_Owner, _Foe, _Stack)
THEN
NOT DB_ARCANA_DoubtStep(_Owner, _Foe, _Stack);
PROC_ARCANA_Aspect_Step(_Owner, "EMPRESS");
//END_REGION

//REGION Tier II: Two Pillars (the Deluded enemy's nearest unmarked neighbour within 6 m gains her Sigil)
IF
StatusApplied((CHARACTER)_Foe, "DELUDED", (CHARACTER)_Deceiver, _)
THEN
PROC_ARCANA_Empress_Chain(_Deceiver, _Foe);

PROC
PROC_ARCANA_Empress_Chain((CHARACTER)_Deceiver, (CHARACTER)_From)
AND
DB_ARCANA_AspectReached(_Deceiver, 2)
AND
NOT DB_ARCANA_EmpressChained(_Deceiver)
AND
DB_ARCANA_Aspect_InCombat(_Deceiver, _Combat)
THEN
PROC_ARCANA_Empress_ClearBest(_Deceiver);
PROC_ARCANA_Empress_Consider(_Deceiver, _From, _Combat);
PROC_ARCANA_Empress_Mark(_Deceiver);

PROC
PROC_ARCANA_Empress_ClearBest((CHARACTER)_Deceiver)
AND
DB_ARCANA_EmpressBest(_Deceiver, _Cand, _Dist)
THEN
NOT DB_ARCANA_EmpressBest(_Deceiver, _Cand, _Dist);

PROC
PROC_ARCANA_Empress_Consider((CHARACTER)_Deceiver, (CHARACTER)_From, (GUIDSTRING)_Combat)
AND
DB_ARCANA_Aspect_InCombat(_Cand, _Combat)
AND
_Cand != _From
AND
IsEnemy(_Deceiver, _Cand, 1)
AND
IsDead(_Cand, 0)
AND
HasActiveStatus(_Cand, "SIGIL", 0)
AND
GetDistanceTo(_From, _Cand, _Dist)
AND
_Dist <= 6.0
THEN
PROC_ARCANA_Empress_Offer(_Deceiver, _Cand, _Dist);

PROC
PROC_ARCANA_Empress_Offer((CHARACTER)_Deceiver, (CHARACTER)_Cand, (REAL)_Dist)
AND
DB_ARCANA_EmpressBest(_Deceiver, _Old, _OldDist)
AND
_Dist < _OldDist
THEN
NOT DB_ARCANA_EmpressBest(_Deceiver, _Old, _OldDist);
DB_ARCANA_EmpressBest(_Deceiver, _Cand, _Dist);

PROC
PROC_ARCANA_Empress_Offer((CHARACTER)_Deceiver, (CHARACTER)_Cand, (REAL)_Dist)
AND
NOT DB_ARCANA_EmpressBest(_Deceiver, _, _)
THEN
DB_ARCANA_EmpressBest(_Deceiver, _Cand, _Dist);

PROC
PROC_ARCANA_Empress_Mark((CHARACTER)_Deceiver)
AND
DB_ARCANA_EmpressBest(_Deceiver, _Cand, _Dist)
THEN
NOT DB_ARCANA_EmpressBest(_Deceiver, _Cand, _Dist);
DB_ARCANA_EmpressChained(_Deceiver);
ApplyStatus(_Cand, "SIGIL", -6.0, 1, _Deceiver);

IF
TurnStarted((CHARACTER)_Deceiver)
AND
DB_ARCANA_EmpressChained(_Deceiver)
THEN
NOT DB_ARCANA_EmpressChained(_Deceiver);
//END_REGION

//REGION Tier III: Which One? (Interrupt_Arcana_Aspect_WhichOne applies WHICH_ONE_TECH on her; the attack already missed)
// She swaps places with her nearest Double and that Double is destroyed in her place (the retire path: no Shatter, no
// blame; the attacker's Doubt comes from the interrupt itself).
IF
StatusApplied((CHARACTER)_Owner, "WHICH_ONE_TECH", _, _)
THEN
PROC_ARCANA_WhichOne_ClearBest(_Owner);
PROC_ARCANA_WhichOne_Consider(_Owner);
PROC_ARCANA_WhichOne_Swap(_Owner);

PROC
PROC_ARCANA_WhichOne_ClearBest((CHARACTER)_Owner)
AND
DB_ARCANA_WhichOneBest(_Owner, _Double, _Dist)
THEN
NOT DB_ARCANA_WhichOneBest(_Owner, _Double, _Dist);

PROC
PROC_ARCANA_WhichOne_Consider((CHARACTER)_Owner)
AND
DB_ARCANA_Double(_Double, _Owner, _Order)
AND
NOT DB_ARCANA_DoubleGone(_Double, _Owner)
AND
IsDead(_Double, 0)
AND
GetDistanceTo(_Owner, _Double, _Dist)
THEN
PROC_ARCANA_WhichOne_Offer(_Owner, _Double, _Dist);

PROC
PROC_ARCANA_WhichOne_Offer((CHARACTER)_Owner, (CHARACTER)_Double, (REAL)_Dist)
AND
DB_ARCANA_WhichOneBest(_Owner, _Old, _OldDist)
AND
_Dist < _OldDist
THEN
NOT DB_ARCANA_WhichOneBest(_Owner, _Old, _OldDist);
DB_ARCANA_WhichOneBest(_Owner, _Double, _Dist);

PROC
PROC_ARCANA_WhichOne_Offer((CHARACTER)_Owner, (CHARACTER)_Double, (REAL)_Dist)
AND
NOT DB_ARCANA_WhichOneBest(_Owner, _, _)
THEN
DB_ARCANA_WhichOneBest(_Owner, _Double, _Dist);

PROC
PROC_ARCANA_WhichOne_Swap((CHARACTER)_Owner)
AND
DB_ARCANA_WhichOneBest(_Owner, _Double, _Dist)
AND
GetPosition(_Owner, _OX, _OY, _OZ)
AND
GetPosition(_Double, _DX, _DY, _DZ)
THEN
NOT DB_ARCANA_WhichOneBest(_Owner, _Double, _Dist);
TeleportToPosition(_Owner, _DX, _DY, _DZ, "ARCANA_WhichOne", 0, 0, 0, 0, 1);
TeleportToPosition(_Double, _OX, _OY, _OZ, "ARCANA_WhichOne", 0, 0, 0, 0, 1);
PROC_ARCANA_Doubles_Retire(_Double, _Owner);
//END_REGION

//REGION Aspect card: The Veil Lifts
// Every Double she has shatters (each casts its own Shatter: 1d6 Psychic and Doubt around it).
IF
CastedSpell((CHARACTER)_Owner, _Spell, _, _, _)
AND
DB_ARCANA_CardFamily(_Spell, "Shout_Arcana_Created_Aspect_VeilLifts")
AND
DB_ARCANA_Double(_Double, _Owner, _Order)
AND
NOT DB_ARCANA_DoubleGone(_Double, _Owner)
AND
IsDead(_Double, 0)
THEN
UseSpell(_Double, "Shout_Deceiver_Double_Shatter", _Double, _Double, 1);

// Each enemy it touched (VEIL_LIFTS_TECH): after the Sigils' and the Shatters' Doubt has landed, Doubt goes to 3, so the
// enemy becomes Deluded through the Doubt goal's own path (stacks reset, steps counted).
IF
StatusApplied((CHARACTER)_Foe, "VEIL_LIFTS_TECH", (CHARACTER)_Owner, _)
THEN
DB_ARCANA_VeilLiftsBy(_Foe, _Owner);
RealtimeObjectTimerCancel(_Foe, "ARCANA_VeilLifts");
RealtimeObjectTimerLaunch(_Foe, "ARCANA_VeilLifts", 900);

IF
ObjectTimerFinished((CHARACTER)_Foe, "ARCANA_VeilLifts")
AND
DB_ARCANA_VeilLiftsBy(_Foe, _Owner)
THEN
NOT DB_ARCANA_VeilLiftsBy(_Foe, _Owner);
PROC_ARCANA_VeilLifts_Delude(_Foe, _Owner);

PROC
PROC_ARCANA_VeilLifts_Delude((CHARACTER)_Foe, (CHARACTER)_Owner)
AND
IsDead(_Foe, 0)
AND
DB_ARCANA_DoubtCount(_Foe, 2)
THEN
PROC_ARCANA_Doubt_Raise(_Foe, _Owner);

PROC
PROC_ARCANA_VeilLifts_Delude((CHARACTER)_Foe, (CHARACTER)_Owner)
AND
IsDead(_Foe, 0)
AND
DB_ARCANA_DoubtCount(_Foe, 1)
THEN
PROC_ARCANA_Doubt_Raise(_Foe, _Owner);
PROC_ARCANA_Doubt_Raise(_Foe, _Owner);
//END_REGION

//REGION Cleanup
PROC
PROC_ARCANA_Aspect_Reset((CHARACTER)_Char)
AND
DB_ARCANA_EmpressChained(_Char)
THEN
NOT DB_ARCANA_EmpressChained(_Char);

PROC
PROC_ARCANA_Aspect_Reset((CHARACTER)_Char)
AND
DB_ARCANA_WhichOneBest(_Char, _Double, _Dist)
THEN
NOT DB_ARCANA_WhichOneBest(_Char, _Double, _Dist);

PROC
PROC_ARCANA_Aspect_Reset((CHARACTER)_Char)
AND
DB_ARCANA_VeilLiftsBy(_Foe, _Char)
THEN
NOT DB_ARCANA_VeilLiftsBy(_Foe, _Char);
//END_REGION
EXITSECTION

ENDEXITSECTION
