KBSECTION
// ARC-0026 stage 1 probes (throwaway; stage 2 deletes this goal and runs RemoveSpell / RemovePassive on them).
// P3: TurnStarted / TurnEnded on a player; P7: AttackedBy from an ally marks in the caster's name, != on CHARACTER;
// P8: count the marks (DB mirror, IsDead), IntegerMin cap 2, show it as the turns of Probe Count.
// Anchors: TurnStarted, StatusApplied / StatusRemoved, IsEnemy, IntegerSum, IntegerMin (mod goals); TurnEnded,
// AttackedBy, AddPassive (vanilla goals).

//REGION Grant the probes
IF
SavegameLoaded()
AND
DB_Players(_Char)
THEN
AddSpell(_Char, "Target_Arcana_Util_PROBE_Mark", 1, 0);
AddSpell(_Char, "Target_Arcana_Util_PROBE_StormPush", 1, 0);
AddSpell(_Char, "Shout_Arcana_Util_PROBE_OpenPath", 1, 0);
AddSpell(_Char, "Shout_Arcana_Util_PROBE_Count", 1, 0);
AddSpell(_Char, "Rush_Arcana_Util_PROBE_Rush", 1, 0);
AddSpell(_Char, "Rush_Arcana_Util_PROBE_Leap", 1, 0);
AddSpell(_Char, "Projectile_Arcana_Util_PROBE_Vortex", 1, 0);
AddPassive(_Char, "PROBE_ARC0026_Touch");
//END_REGION

//REGION Marks mirrored
IF
StatusApplied((CHARACTER)_Foe, "PROBE_ARC0026_MARK", (CHARACTER)_Src, _)
THEN
DB_ARCANA_ProbeMark(_Src, _Foe);

IF
StatusRemoved((CHARACTER)_Foe, "PROBE_ARC0026_MARK", _, _)
AND
DB_ARCANA_ProbeMark(_Src, _Foe)
THEN
NOT DB_ARCANA_ProbeMark(_Src, _Foe);

// P7: while a player has Probe Open Path, an ally's hit marks the enemy in that player's name
IF
AttackedBy((CHARACTER)_Foe, (CHARACTER)_Ally, _, _, _, _, _)
AND
DB_Players(_Src)
AND
HasActiveStatus(_Src, "PROBE_ARC0026_PATH", 1)
AND
_Ally != _Src
AND
IsEnemy(_Src, _Foe, 1)
AND
IsEnemy(_Src, _Ally, 0)
THEN
ApplyStatus(_Foe, "PROBE_ARC0026_MARK", 12.0, 1, _Src);
//END_REGION

//REGION P8 count, P3 sweep and turn end
IF
CastedSpell((CHARACTER)_Src, "Shout_Arcana_Util_PROBE_Count", _, _, _)
THEN
PROC_ARCANA_Probe26_Count(_Src);
PROC_ARCANA_Probe26_Show(_Src);

PROC
PROC_ARCANA_Probe26_Count((CHARACTER)_Src)
AND
DB_ARCANA_Probe26X(_Src, _Old)
THEN
NOT DB_ARCANA_Probe26X(_Src, _Old);

PROC
PROC_ARCANA_Probe26_Count((CHARACTER)_Src)
THEN
DB_ARCANA_Probe26X(_Src, 0);

PROC
PROC_ARCANA_Probe26_Count((CHARACTER)_Src)
AND
DB_ARCANA_ProbeMark(_Src, _Foe)
AND
IsDead(_Foe, 0)
AND
DB_ARCANA_Probe26X(_Src, _N)
AND
IntegerSum(_N, 1, _M)
THEN
NOT DB_ARCANA_Probe26X(_Src, _N);
DB_ARCANA_Probe26X(_Src, _M);

PROC
PROC_ARCANA_Probe26_Show((CHARACTER)_Src)
AND
DB_ARCANA_Probe26X(_Src, _X)
AND
IntegerMin(_X, 2, _T)
AND
_T == 1
THEN
ApplyStatus(_Src, "PROBE_ARC0026_COUNT", 6.0, 1, _Src);

PROC
PROC_ARCANA_Probe26_Show((CHARACTER)_Src)
AND
DB_ARCANA_Probe26X(_Src, _X)
AND
IntegerMin(_X, 2, _T)
AND
_T == 2
THEN
ApplyStatus(_Src, "PROBE_ARC0026_COUNT", 12.0, 1, _Src);

IF
TurnStarted((CHARACTER)_Src)
AND
HasActiveStatus(_Src, "PROBE_ARC0026_COUNT", 1)
THEN
PROC_ARCANA_Probe26_Sweep(_Src);

PROC
PROC_ARCANA_Probe26_Sweep((CHARACTER)_Src)
AND
DB_ARCANA_ProbeMark(_Src, _Foe)
THEN
RemoveStatus(_Foe, "PROBE_ARC0026_MARK", _Src);

IF
TurnEnded((CHARACTER)_Src)
AND
DB_Players(_Src)
AND
HasActiveStatus(_Src, "PROBE_ARC0026_COUNT", 1)
THEN
ApplyStatus(_Src, "PROBE_ARC0026_ENDED", 6.0, 1, _Src);
//END_REGION
EXITSECTION

ENDEXITSECTION
