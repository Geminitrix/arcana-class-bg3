KBSECTION
//REGION Limpeza compartilhada
PROC
PROC_ARCANA_MirrorWithdraw_ClearPosition((CHARACTER)_Caster)
AND
DB_MirrorWithdraw_Position(_Caster, _X, _Y, _Z)
THEN
NOT DB_MirrorWithdraw_Position(_Caster, _X, _Y, _Z);

PROC
PROC_ARCANA_MirrorWithdraw_ClearMarker((CHARACTER)_Caster)
AND
DB_MirrorWithdraw_Marker(_Caster, _Marker)
THEN
RequestDelete((ITEM)_Marker);
NOT DB_MirrorWithdraw_Marker(_Caster, _Marker);

PROC
PROC_ARCANA_MirrorWithdraw_Clear((CHARACTER)_Caster)
THEN
ObjectTimerCancel(_Caster, "ARCANA_MirrorWithdraw_Timer");
RemoveSpell(_Caster, "Shout_Deceiver_Clone_MirrorWithdraw", 0);
PROC_ARCANA_MirrorWithdraw_ClearPosition(_Caster);
PROC_ARCANA_MirrorWithdraw_ClearMarker(_Caster);
//END_REGION

//REGION Lancou o Mirror Distortion - guarda posicao e abre a janela
IF
UsingSpell((CHARACTER)_Caster, "Target_Deceiver_Clone_MirrorDistortion", _, _, _)
AND
GetPosition(_Caster, _X, _Y, _Z)
THEN
DB_MirrorWithdraw_Position(_Caster, _X, _Y, _Z);
AddSpell(_Caster, "Shout_Deceiver_Clone_MirrorWithdraw", 1, 0);
ObjectTimerCancel(_Caster, "ARCANA_MirrorWithdraw_Timer");
ObjectTimerLaunch(_Caster, "ARCANA_MirrorWithdraw_Timer", 12000, 1);

IF
UsingSpell((CHARACTER)_Caster, "Target_Deceiver_Clone_MirrorDistortion", _, _, _)
THEN
ApplyStatus(_Caster, "MIRROR_DISTORTION_HELPER", 12.0, 1, _Caster);

IF
EnteredLevel((ITEM)_Marker, 4441890b-a8c1-4c6d-8309-ec505adc7acc, _)
AND
GetPosition(_Marker, _MX, _MY, _MZ)
AND
DB_MirrorWithdraw_Position(_Caster, _X, _Y, _Z)
AND
GetDistanceToPosition(_Marker, _X, _Y, _Z, _Dist)
AND
_Dist < 1.0
THEN
DB_MirrorWithdraw_Marker(_Caster, _Marker);
//END_REGION

//REGION A janela expirou
IF
ObjectTimerFinished((CHARACTER)_Caster, "ARCANA_MirrorWithdraw_Timer")
THEN
PROC_ARCANA_MirrorWithdraw_Clear(_Caster);
//END_REGION

//REGION Usou o Mirror Withdraw - teleporta e encerra
IF
CastSpell((CHARACTER)_Caster, "Shout_Deceiver_Clone_MirrorWithdraw", _, _, _)
AND
DB_MirrorWithdraw_Position(_Caster, _X, _Y, _Z)
THEN
AppearAtPosition(_Caster, _X, _Y, _Z, 0, CUST_MizoraTeleIn_01_99069369-88a0-4580-84ab-aa34788cb509, "Created1", 1);
PROC_ARCANA_MirrorWithdraw_Clear(_Caster);
//END_REGION
EXITSECTION

ENDEXITSECTION
